cmake_minimum_required(VERSION 3.16)
project(consumer LANGUAGES CXX)
find_package(sampler CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE sampler::sampler)
get_target_property(compiled_kind sampler::sampler TYPE)
get_target_property(utility_kind sampler::sample_constants TYPE)
if(NOT compiled_kind STREQUAL "STATIC_LIBRARY" OR NOT utility_kind STREQUAL "INTERFACE_LIBRARY")
  message(FATAL_ERROR "Expected compiled STATIC and header-only INTERFACE targets")
endif()
add_executable(utility_consumer utility.cpp)
target_link_libraries(utility_consumer PRIVATE sampler::sample_constants)
