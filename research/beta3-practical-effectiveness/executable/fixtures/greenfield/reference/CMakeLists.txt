cmake_minimum_required(VERSION 3.16)
project(sampler LANGUAGES CXX)
add_library(sample_constants INTERFACE)
target_compile_features(sample_constants INTERFACE cxx_std_17)
target_include_directories(sample_constants INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<INSTALL_INTERFACE:include>)
add_library(sampler STATIC src/demo/audioPlayer.cpp)
target_compile_features(sampler PUBLIC cxx_std_17)
target_include_directories(sampler PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<INSTALL_INTERFACE:include>)
target_link_libraries(sampler PUBLIC sample_constants)
install(TARGETS sampler sample_constants EXPORT samplerTargets ARCHIVE DESTINATION lib)
install(DIRECTORY src/ DESTINATION include FILES_MATCHING PATTERN "*.hpp")
install(EXPORT samplerTargets NAMESPACE sampler:: FILE samplerConfig.cmake DESTINATION lib/cmake/sampler)
